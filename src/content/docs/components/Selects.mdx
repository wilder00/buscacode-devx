import React from 'react'
import { Select } from '@buscacode/selects'
import { EyeOn } from '@buscacode/icons'
import ShowArea from '@/components/astro/ShowArea.astro'

Selects to be used with buscacode

## Simple use

<ShowArea title="algo">
  <Select label="Label" leading={React.createElement(EyeOn)}>
    <option value="A">opción 1</option>
    <option value="B">opción 2</option>
    <option value="C">opción 3</option>
  </Select>
</ShowArea>

```tsx
<Select label="Label" leading={<EyeOn />}>
  <option value="A">opción 1</option>
  <option value="B">opción 2</option>
  <option value="C">opción 3</option>
</Select>
```
