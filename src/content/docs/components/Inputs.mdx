import React from 'react'
import { Input } from '@buscacode/inputs'
import { EyeOn } from '@buscacode/icons'
import ShowArea from '@/components/astro/ShowArea.astro'

Inputs to be used with buscacode

## Simple use

<ShowArea title="algo">
  <Input
    label="Label"
    leading={React.createElement(EyeOn)}
    placeholder="Password"
  />
</ShowArea>

```tsx
<Input label="Label" leading={<EyeOn />} placeholder="Password" />
```

## Modifying leading icon colors

<ShowArea title="algo">
  <Input
    value="Hello World"
    leading={React.createElement(
      'span',
      {
        className:
          'text-bc-primary-50 dark:text-bc-neutral-400 flex items-center justify-center'
      },
      React.createElement(EyeOn)
    )}
  />
</ShowArea>

```tsx
<Input
  leading={
    <span className="flex items-center justify-center text-bc-primary-50 dark:text-bc-neutral-400">
      <EyeOn />
    </span>
  }
/>
```
